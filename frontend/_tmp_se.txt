21|function distPointSeg(p: Point2D, a: Point2D, b: Point2D): { dist: number; t: number } {
22|  const dx = b.x - a.x
23|  const dy = b.y - a.y
24|  const lenSq = dx * dx + dy * dy
25|  if (lenSq < 1e-9) return { dist: Math.hypot(p.x - a.x, p.y - a.y), t: 0 }
26|  const t = Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / lenSq))
27|  return { dist: Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy)), t }

30|/**
31| * Surface polygoon-edit: sleep vertices (andere polygonen hoek/ribbe + H/V,
32| * Shift = ribben H/V, Ctrl/Cmd = uit), klik op rand → punt invoegen.
33| */
34|export function usePlanCanvasSurfaceEdit(options: {
35|  selection: PlanCanvasSelectionRefs
36|  editor: EditorApi
37|  hitTest: Pick<HitTestApi, 'clientToCm'> & Partial<Pick<HitTestApi, 'handleHitTolCm'>>
38|  resolvePoint: (
39|    cm: Point2D,
40|    snapDisabled: boolean,
41|    extraAxisPoints?: Point2D[],
42|    excludeSurfaceId?: string | null,
43|  ) => Point2D
44|  axisLocked: { value: boolean }
45|  syncPlanToParent: () => void
46|  /** Nokbalk onder de klik: niet opeten, zodat de nok geopend kan worden. */
47|  isRidgeHit?: (cm: Point2D) => boolean
48|}) {
49|  const draggingVertexIndex = ref<number | null>(null)
50|  const selectedVertexIndex = ref<number | null>(null)
51|
52|  watch(
53|    () => options.selection.surfaceEditId.value,
54|    () => {
55|      selectedVertexIndex.value = null
56|    },
57|  )
58|  let didPushUndo = false
59|  let snapDisabled = false
60|  let pendingZ: { index: number; z: number } | null = null
61|  let dragMoveHandler: ((event: MouseEvent) => void) | null = null
62|  let bodyDrag: { startCm: Point2D; startPoly: Point2D[] } | null = null
63|  let roofDragStartPoly: Array<Point2D & { z?: number }> | null = null
64|  const DRAG_START_PX = 4
65|
66|  function captureRoofDragStart(): void {
67|    const poly = currentSurface()?.poly
68|    roofDragStartPoly = poly?.map((p) => ({ x: p.x, y: p.y, z: p.z })) ?? null
69|  }
70|
71|  function flushRoofWallFollow(): void {
72|    const id = options.selection.surfaceEditId.value
73|    if (!id || !isRoofSurface(currentSurface())) {
74|      roofDragStartPoly = null
75|      return
76|    }
77|    options.editor.applyWallsAfterRoofEdit(id, roofDragStartPoly ?? undefined)
78|    roofDragStartPoly = null
79|  }
80|
81|  function isEditing(): boolean {
82|    return options.selection.surfaceEditId.value != null
83|  }
84|
85|  function currentSurface(): FloorSurface | undefined {
86|    const id = options.selection.surfaceEditId.value
87|    if (!id) return undefined
88|    return options.editor.surfaces.value.find((s) => s.id === id)
89|  }
90|
91|  function currentPoly(): Array<Point2D & { z?: number }> | null {
92|    const surface = currentSurface()
93|    if (!surface?.poly?.length) return null
94|    return surface.poly.map((p) => ({ x: p.x, y: p.y, z: p.z }))
95|  }
96|
97|  function resolveVertexZ(point: Point2D, previous?: { x: number; y: number; z?: number }): number {
98|    if (previous && Math.hypot(previous.x - point.x, previous.y - point.y) < 0.05) {
99|      return previous.z ?? 0
100|    }
101|    const plan = options.editor.localPlan.value
102|    if (plan && isRoofSurface(currentSurface())) {
103|      return snapRoofVertexZ({
104|        plan,
105|        floorIndex: options.editor.floorIndex.value,
106|        point,
107|      })
108|    }
109|    return previous?.z ?? 0
110|  }
111|
112|  function commitPoly(poly: Array<Point2D & { z?: number }>): void {
113|    const id = options.selection.surfaceEditId.value
114|    if (!id) return
115|    if (!didPushUndo) {
116|      options.editor.pushUndo()
117|      didPushUndo = true
118|    }
119|    const existing = currentSurface()
120|    const dragIdx = draggingVertexIndex.value
121|    options.editor.updateSurface(id, {
122|      poly: poly.map((p, index) => ({
123|        x: p.x,
124|        y: p.y,
125|        z:
126|          index === dragIdx
127|            ? resolveVertexZ(p, existing?.poly[index])
128|            : (p.z ?? existing?.poly[index]?.z ?? resolveVertexZ(p)),
129|      })),
130|    })
131|  }
132|
133|  function cleanupDragListeners(): void {
134|    if (dragMoveHandler) window.removeEventListener('pointermove', dragMoveHandler)
135|    dragMoveHandler = null
136|    window.removeEventListener('pointermove', onDragMove)
137|    window.removeEventListener('pointerup', onDragUp)
138|  }
139|
140|  function resolveEventPoint(event: MouseEvent): Point2D | null {
141|    const cm = options.hitTest.clientToCm(event.clientX, event.clientY)
142|    if (!cm) return null
143|    const disabled = snapDisabled || event.ctrlKey || event.metaKey
144|    const poly = currentPoly()
145|    const idx = draggingVertexIndex.value
146|    const extra = poly && idx != null ? poly.filter((_, i) => i !== idx) : undefined
147|    let point = options.resolvePoint(cm, disabled, extra, options.selection.surfaceEditId.value)
148|    if (options.axisLocked.value && poly && idx != null) {
149|      point = snapPolygonVertexAxisLock(poly, idx, point)
150|    }
151|    return point
152|  }
153|
154|  function onDragMove(event: MouseEvent): void {
155|    if (draggingVertexIndex.value == null) return
156|    snapDisabled = snapDisabled || event.ctrlKey || event.metaKey
157|    const point = resolveEventPoint(event)
158|    if (!point) return
159|    const poly = currentPoly()
160|    if (!poly) return
161|    const idx = draggingVertexIndex.value
162|    if (idx < 0 || idx >= poly.length) return
163|    const next = poly.map((p, i) => (i === idx ? { x: point.x, y: point.y } : p))
164|    commitPoly(next)
165|  }
166|
167|  function onDragUp(): void {
168|    draggingVertexIndex.value = null
169|    bodyDrag = null
170|    snapDisabled = false
171|    cleanupDragListeners()
172|    if (didPushUndo) {
173|      flushRoofWallFollow()
174|      options.syncPlanToParent()
175|      didPushUndo = false
176|    } else {
177|      roofDragStartPoly = null
178|    }
179|  }
180|
181|  function beginVertexDrag(index: number, event: MouseEvent): void {
182|    const startX = event.clientX
183|    const startY = event.clientY
184|    draggingVertexIndex.value = null
185|    didPushUndo = false
186|    captureRoofDragStart()
187|    snapDisabled = event.ctrlKey || event.metaKey
188|    const onMove = (move: MouseEvent) => {
189|      if (draggingVertexIndex.value == null) {
190|        if (Math.hypot(move.clientX - startX, move.clientY - startY) < DRAG_START_PX) return
191|        draggingVertexIndex.value = index
192|      }
193|      onDragMove(move)
194|    }
195|    dragMoveHandler = onMove
196|    const onUp = () => {
197|      window.removeEventListener('pointermove', onMove)
198|      if (dragMoveHandler === onMove) dragMoveHandler = null
199|      onDragUp()
200|    }
201|    window.addEventListener('pointermove', onMove)
202|    window.addEventListener('pointerup', onUp, { once: true })
203|  }
204|
205|  function beginBodyDrag(event: MouseEvent, startCm: Point2D, startPoly: Point2D[]): void {
206|    const startX = event.clientX
207|    const startY = event.clientY
208|    bodyDrag = null
209|    didPushUndo = false
210|    captureRoofDragStart()
211|    const onMove = (move: MouseEvent) => {
212|      const cm = options.hitTest.clientToCm(move.clientX, move.clientY)
213|      if (!cm) return
214|      if (bodyDrag == null) {
215|        if (Math.hypot(move.clientX - startX, move.clientY - startY) < DRAG_START_PX) return
216|        bodyDrag = { startCm, startPoly }
217|      }
218|      const dx = cm.x - bodyDrag.startCm.x
219|      const dy = cm.y - bodyDrag.startCm.y
220|      commitPoly(bodyDrag.startPoly.map((p) => ({ x: p.x + dx, y: p.y + dy })))
221|    }
222|    dragMoveHandler = onMove
223|    const onUp = () => {
224|      window.removeEventListener('pointermove', onMove)
225|      if (dragMoveHandler === onMove) dragMoveHandler = null
226|      onDragUp()
227|    }
228|    window.addEventListener('pointermove', onMove)
229|    window.addEventListener('pointerup', onUp, { once: true })
230|  }

232|  /** @returns true when the event was consumed by edit mode. */
233|  function removeVertex(index: number): void {
234|    const poly = currentPoly()
235|    if (!poly || poly.length <= 3) return
236|    if (index < 0 || index >= poly.length) return
237|    const next = poly.filter((_, i) => i !== index)
238|    commitPoly(next)
239|    selectedVertexIndex.value = Math.min(index, next.length - 1)
240|    options.syncPlanToParent()
241|    didPushUndo = false
242|  }
243|
244|  function onPointerDown(event: MouseEvent): boolean {
245|    flushPendingVertexZ()
246|    if (!isEditing()) return false
247|    const raw = options.hitTest.clientToCm(event.clientX, event.clientY)
248|    if (!raw) return false
249|    const poly = currentPoly()
250|    if (!poly || poly.length < 3) return false
251|    const mutate = options.selection.roofPolyMutate.value === true || event.ctrlKey || event.metaKey
252|    const tol = options.hitTest.handleHitTolCm?.() ?? VERTEX_HIT_FALLBACK_CM
253|    const hitVi = hitSelectedVertex(poly, raw, tol, selectedVertexIndex.value)
254|    if (hitVi != null) {
255|      if ((event.ctrlKey || event.metaKey) && poly.length > 3) {
256|        removeVertex(hitVi)
257|        event.preventDefault()
258|        return true
259|      }
260|      selectedVertexIndex.value = hitVi
261|      beginVertexDrag(hitVi, event)
262|      event.preventDefault()
263|      return true
264|    }
265|
266|    let bestEi = -1
267|    let bestEd = Number.POSITIVE_INFINITY
268|    let bestT = 0.5
269|    for (let i = 0; i < poly.length; i++) {
270|      const a = poly[i]
271|      const b = poly[(i + 1) % poly.length]
272|      const { dist, t } = distPointSeg(raw, a, b)
273|      if (dist < bestEd) {
274|        bestEd = dist
275|        bestEi = i
276|        bestT = t
277|      }
278|    }
279|    if (mutate && bestEi >= 0 && bestEd <= EDGE_HIT_CM && bestT > 0.05 && bestT < 0.95) {
280|      const a = poly[bestEi]
281|      const b = poly[(bestEi + 1) % poly.length]
282|      const along = {
283|        x: a.x + (b.x - a.x) * bestT,
284|        y: a.y + (b.y - a.y) * bestT,
285|      }
286|      const snapDisabledNow = event.ctrlKey || event.metaKey
287|      const inserted = options.resolvePoint(
288|        along,
289|        snapDisabledNow,
290|        poly,
291|        options.selection.surfaceEditId.value,
292|      )
293|      const plan = options.editor.localPlan.value
294|      const insertedZ =
295|        plan && isRoofSurface(currentSurface())
296|          ? resolveInsertedRoofVertexZ({
297|              plan,
298|              floorIndex: options.editor.floorIndex.value,
299|              point: inserted,
300|              edgeA: currentSurface()?.poly[bestEi],
301|              edgeB: currentSurface()?.poly[(bestEi + 1) % poly.length],
302|              t: bestT,
303|            })
304|          : (currentSurface()?.poly[bestEi]?.z ?? 0)
305|      const next = [
306|        ...poly.slice(0, bestEi + 1),
307|        { ...inserted, z: insertedZ },
308|        ...poly.slice(bestEi + 1),
309|      ]
310|      commitPoly(next)
311|      selectedVertexIndex.value = bestEi + 1
312|      beginVertexDrag(bestEi + 1, event)
313|      event.preventDefault()
314|      return true
315|    }
316|
317|    if (options.isRidgeHit?.(raw) === true) return false
318|    if (pointInPoly(raw, poly)) {
319|      if (isRoofSurface(currentSurface())) {
320|        beginBodyDrag(event, raw, poly)
321|      }
322|      event.preventDefault()
323|      return true
324|    }
325|
326|    return false
327|  }
328|
329|  function cancelDrag(): void {
330|    cleanupDragListeners()
331|    draggingVertexIndex.value = null
332|    bodyDrag = null
333|    snapDisabled = false
334|    didPushUndo = false
335|    roofDragStartPoly = null
336|  }
337|
338|  function applyVertexZ(index: number, zCm: number): void {
339|    const surface = currentSurface()
340|    if (!surface?.poly[index]) return
341|    const plan = options.editor.localPlan.value
342|    const z = isRoofSurface(surface)
343|      ? clampRoofVertexZCm(
344|          zCm,
345|          plan ? slabCmForRoofSurface(plan, surface.id) : DEFAULT_FLOOR_THICKNESS_CM,
346|        )
347|      : Math.max(0, Math.round(zCm))
348|    if (Math.round(surface.poly[index]?.z ?? 0) === z) return
349|    if (!didPushUndo) {
350|      options.editor.pushUndo()
351|      didPushUndo = true
352|    }
353|    const next = surface.poly.map((point, i) => (i === index ? { ...point, z } : point))
354|    options.editor.updateSurface(surface.id, { poly: next })
355|    if (isRoofSurface(surface)) {
356|      options.editor.applyWallsAfterRoofEdit(surface.id)
357|    }
358|    options.syncPlanToParent()
359|    didPushUndo = false
360|  }
361|
362|  function flushPendingVertexZ(): void {
363|    if (!pendingZ) return
364|    applyVertexZ(pendingZ.index, pendingZ.z)
365|    pendingZ = null
366|  }
367|
368|  function setSelectedVertexZ(zCm: number): void {
369|    const idx = selectedVertexIndex.value
370|    if (idx == null) return
371|    pendingZ = { index: idx, z: zCm }
372|    applyVertexZ(idx, pendingZ.z)
373|  }
374|
375|  return {
376|    isEditing,
377|    onPointerDown,
378|    cancelDrag,
379|    draggingVertexIndex,
380|    selectedVertexIndex,
381|    setSelectedVertexZ,
382|  }

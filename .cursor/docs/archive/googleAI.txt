// En dat de tekenaar de coördinaten van een deur-voorbeeld heeft gegeven

function verwerkBouwtekening(src, deurVoorbeeldCoords, muurDiktePixels) {
    // 1. INITIALISATIE VAN DE MATRICES
    let graySrc = new cv.Mat();
    cv.cvtColor(src, graySrc, cv.col_code.COLOR_RGBA2GRAY); // Omzetten naar zwart-wit voor CV

    // 2. TEMPLATE MATCHING (Deuren & Ramen)
    // Snijd het deursymbool uit de tekening op basis van de selectie van de tekenaar
    let rect = new cv.Rect(deurVoorbeeldCoords.x, deurVoorbeeldCoords.y, deurVoorbeeldCoords.w, deurVoorbeeldCoords.h);
    let template = graySrc.roi(rect);

    let matchResult = new cv.Mat();
    let mask = new cv.Mat.zeros(graySrc.rows, graySrc.cols, cv.CV_8UC1); // Masker om deuren 'wit te gummen'

    // Voer de template matching uit
    cv.matchTemplate(graySrc, template, matchResult, cv.TM_CCOEFF_NORMED);
    
    // Loop door de resultaten om hits te vinden (boven 85% overeenkomst)
    let dmatch = new cv.Mat();
    let threshold = 0.85;
    
    // In een productie-omgeving zou u hier ook rotaties van het template herhalen (90, 180, 270 graden)
    for (let r = 0; r < matchResult.rows; r++) {
        for (let c = 0; c < matchResult.cols; c++) {
            if (matchResult.floatAt(r, c) > threshold) {
                // We hebben een deur gevonden! 
                // Sla de coördinaten op voor de JSON output
                let gevondenDeur = { x: c, y: r, w: deurVoorbeeldCoords.w, h: deurVoorbeeldCoords.h };
                
                // STAP A: WIGGUMMEN (Masking) van de deur voor de muurdetectie
                let deurRegio = new cv.Rect(c, r, deurVoorbeeldCoords.w, deurVoorbeeldCoords.h);
                // We maken de regio wit (255) op ons masker
                let color = new cv.Scalar(255, 255, 255, 255);
                cv.rectangle(mask, new cv.Point(c, r), new cv.Point(c + deurVoorbeeldCoords.w, r + deurVoorbeeldCoords.h), color, -1);
            }
        }
    }

    // Pas het masker toe op de originele tekening: deuren worden nu wit overschreven
    let cleanSrcForWalls = graySrc.clone();
    cleanSrcForWalls.setTo(new cv.Scalar(255), mask); // Deuren zijn nu 'weggegumd'

    // 3. MUURDETECTIE (Filteren op dikte & Lijnherkenning)
    // STAP B: Filteren op dikte d.m.v. Morfologische operaties (Erode & Dilate)
    // Dit verwijdert dunne maatlijnen en vouwlijnen
    let filteredWalls = new cv.Mat();
    let M = cv.Mat.ones(muurDiktePixels, muurDiktePixels, cv.CV_8U); // Kernel ter grootte van de muurdikte
    
    // Erodeer (maak lijnen dunner, dunne lijnen verdwijnen volledig) en dilateer (maak weer originele dikte)
    cv.erode(cleanSrcForWalls, filteredWalls, M, new cv.Point(-1, -1), 1, cv.BORDER_CONSTANT, cv.morphologyDefaultBorderValue());
    cv.dilate(filteredWalls, filteredWalls, M, new cv.Point(-1, -1), 1, cv.BORDER_CONSTANT, cv.morphologyDefaultBorderValue());

    // STAP C: Gaten dichten met Probabilistic Hough Lines
    let lines = new cv.Mat();
    // Parameter uitleg: 
    // threshold=50 (minimaal aantal stemmen), minLineLength=40 pixels, maxLineGap=60 pixels (overbrugt het deurgat)
    cv.HoughLinesP(filteredWalls, lines, 1, Math.PI / 180, 50, 40, 60);

    let muurVectorenJSON = [];

    // Loop door alle gedetecteerde muurlijnen
    for (let i = 0; i < lines.rows; ++i) {
        let x1 = lines.data32S[i * 4];
        let y1 = lines.data32S[i * 4 + 1];
        let x2 = lines.data32S[i * 4 + 2];
        let y2 = lines.data32S[i * 4 + 3];

        // Sla de lijnsegmenten op
        muurVectorenJSON.push({
            start: { x: x1, y: y1 },
            end: { x: x2, y: y2 }
        });
    }

    // STAP D: Hoekpunten oplossen (Wiskundige snapping)
    // (In JavaScript doorloopt u nu 'muurVectorenJSON' om lijnen die nagenoeg 
    // haaks op elkaar staan en elkaar bijna raken, wiskundig naar elkaar toe te trekken)
    let strakkeMuren = losHoekpuntenOp(muurVectorenJSON);

    // Geheugen netjes opruimen (Cruciaal in OpenCV.js/WebAssembly om crashes te voorkomen!)
    graySrc.delete(); template.delete(); matchResult.delete(); 
    mask.delete(); cleanSrcForWalls.delete(); filteredWalls.delete(); 
    M.delete(); lines.delete();

    return strakkeMuren;
}

function losHoekpuntenOp(lijnen) {
    let snappingAfstand = 15; // Maximaal aantal pixels dat een hoek open mag zijn om hem te dichten
    
    for (let i = 0; i < lijnen.length; i++) {
        for (let j = i + 1; j < lijnen.length; j++) {
            // Wiskundige check: liggen de eindpunten van lijn I en lijn J heel dicht bij elkaar?
            // Zo ja, maak de coördinaten exact gelijk aan elkaar (snapping).
            // Dit dicht de L- en T-splitsingen in de hoeken van kamers.
        }
    }
    return lijnen;
}

    // Lijn 1: (x1, y1) naar (x2, y2) -> Formule: A1*x + B1*y = C1
    let A1 = lijn1.end.y - lijn1.start.y;
    let B1 = lijn1.start.x - lijn1.end.x;
    let C1 = A1 * lijn1.start.x + B1 * lijn1.start.y;

    // Lijn 2: (x3, y3) naar (x4, y4) -> Formule: A2*x + B2*y = C2
    let A2 = lijn2.end.y - lijn2.start.y;
    let B2 = lijn2.start.x - lijn2.end.x;
    let C2 = A2 * lijn2.start.x + B2 * lijn2.start.y;

    let det = A1 * B2 - A2 * B1;
    if (det === 0) return null; // Lijnen lopen parallel, geen snijpunt

    // Bereken het exacte X,Y snijpunt
    let x = (B2 * C1 - B1 * C2) / det;
    let y = (A1 * C2 - A2 * C1) / det;
    return { x: x, y: y };
}
